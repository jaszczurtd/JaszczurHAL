# Separate objects exercise C linkage and replacement during the link step.
add_library(weak_function_defaults STATIC defaults.c defaults.cpp)
target_include_directories(weak_function_defaults PRIVATE "${PROJECT_SOURCE_DIR}/src")
jh_enable_host_warnings(weak_function_defaults ERRORS)

foreach(placement IN ITEMS object archive)
    foreach(implementation IN ITEMS default override)
        set(probe test_weak_function_${placement}_${implementation})
        add_executable(${probe} caller.c caller.cpp)
        if(placement STREQUAL "object")
            target_sources(${probe} PRIVATE defaults.c defaults.cpp)
        else()
            target_link_libraries(${probe} PRIVATE weak_function_defaults)
        endif()
        if(implementation STREQUAL "override")
            target_sources(${probe} PRIVATE override.c override.cpp)
            target_compile_definitions(${probe} PRIVATE EXPECTED_OFFSET=100)
        else()
            target_compile_definitions(${probe} PRIVATE EXPECTED_OFFSET=10)
        endif()
        target_include_directories(${probe} PRIVATE "${PROJECT_SOURCE_DIR}/src")
        jh_enable_host_warnings(${probe} ERRORS)
        add_test(NAME ${probe} COMMAND ${probe})
    endforeach()
endforeach()
